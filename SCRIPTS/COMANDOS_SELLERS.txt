  2. DIEGO.EX  (ya lo hizo, por si hay que repetirlo)
=====================================================================

  ARCHIVO: 5000_productos.csv  (el del profesor)

      mysql -u diego.ex -p240709 -h 10.10.154.221 -P 40107 --local-infile=1 db_test

  Bloque LOAD DATA (el mismo que uso, pero con LEFT(sku,3) para la
  categoria, porque ese CSV trae prefijo de 3 letras en el SKU):

      LOAD DATA LOCAL INFILE 'C:/ruta/al/archivo/5000_productos.csv'
      INTO TABLE tb_products
      CHARACTER SET utf8mb4
      FIELDS TERMINATED BY ',' OPTIONALLY ENCLOSED BY '"' ESCAPED BY '"'
      LINES TERMINATED BY '\r\n'
      IGNORE 1 LINES
      (@id, sku, name, description, @precio, @stock, @status, @fecha, @upd)
      SET price        = @precio,
          stock        = @stock,
          status       = IF(@status = '1', b'1', b'0'),
          category     = CASE LEFT(sku, 3)
                            WHEN 'ENE' THEN 'Energia'
                            WHEN 'OFI' THEN 'Oficina'
                            WHEN 'ELE' THEN 'Electronica'
                            WHEN 'AUD' THEN 'Audio'
                            WHEN 'ACC' THEN 'Accesorios'
                            WHEN 'MOV' THEN 'Moviles'
                            WHEN 'HOG' THEN 'Hogar'
                            WHEN 'RED' THEN 'Redes'
                            WHEN 'ALM' THEN 'Almacenamiento'
                            WHEN 'GAM' THEN 'Gaming'
                            ELSE 'Otros'
                          END,
          creation_date = STR_TO_DATE(@fecha, '%d/%m/%Y %H:%i'),
          last_update   = STR_TO_DATE(@upd,  '%d/%m/%Y %H:%i');

      SELECT ROW_COUNT() AS filas_cargadas;
      exit;
